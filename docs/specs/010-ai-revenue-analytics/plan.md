# Implementation Plan: AI Revenue Analytics for Providers

**Branch**: `main` | **Date**: 2026-07-05 | **Spec**: [spec.md](spec.md)

## Summary

Add an AI-powered revenue analysis panel to the Provider Dashboard. An admin toggles the global `AI_REVENUE_ANALYTICS` feature flag via a real backend API (currently mock-only). When enabled, providers click "✨ Phân tích AI" to receive a structured Vietnamese insight report generated by Gemini. Each request is logged in `ai_analysis_logs`; that log also serves as the per-provider monthly quota counter.

## Technical Context

**Language/Version**: TypeScript 5.x strict, Node.js 20+
**Primary Dependencies**: Express.js, TypeORM, PostgreSQL, `@google/genai` (existing), zod
**Gemini model**: `env.ai.supportModel` (gemini-2.0-flash) — cost-efficient, already configured
**Storage**: PostgreSQL — new `ai_analysis_logs` table; existing `feature_flags` table extended with `config` JSONB
**Testing**: Manual E2E (existing convention)
**Performance Goals**: Full AI response within 15 s; quota check + log write under 50 ms
**Constraints**: Must not break existing dashboard or chatbot flows; RBAC enforced at router level

## Architecture Decisions

| Decision | Choice | Rationale |
|----------|--------|-----------|
| Quota tracking | COUNT from `ai_analysis_logs` (status=SUCCESS, current month) | Log table is already required; avoids duplicating state in a separate counter column |
| Feature flag scope | `entity_type='GLOBAL'`, `entity_id=NULL` | v1 is provider-global; per-cafe granularity deferred |
| Monthly quota config | `feature_flags.config->>'monthly_quota'` (0 = unlimited) | Reuses JSONB pattern; readable from admin API |
| HTTP method for AI endpoint | `POST /provider/dashboard/ai-insights` | Side effect: creates a log row — POST is semantically correct |
| Gemini model | `env.ai.supportModel` (gemini-2.0-flash) | Lower cost, sufficient for structured analytics prompts |
| Prompt data | Aggregated totals + time-series from 6 existing service functions | No raw rows; keeps prompt small and avoids PII |
| Admin API surface | `GET /admin/feature-flags` + `PATCH /admin/feature-flags/:key` | Matches existing admin router pattern; minimal — just toggle + config |
| Frontend quota display | Derive from API response `quotaInfo` field | No separate quota endpoint needed; included in POST response |

```typescript
import { Response, NextFunction } from 'express';
import { AuthRequest } from '../types';
import { AppDataSource } from '../config/database';

export const adminFeatureFlagsController = {
  // GET /api/v1/admin/feature-flags
  async list(req: AuthRequest, res: Response, next: NextFunction): Promise<void> {
    try {
      const rows = await AppDataSource.query(
        `SELECT feature_key, entity_type, entity_id, is_enabled, config, updated_at
         FROM feature_flags ORDER BY feature_key`,
      );
      res.json({ success: true, data: rows });
    } catch (err) { next(err); }
  },

  // PATCH /api/v1/admin/feature-flags/:key
  async update(req: AuthRequest, res: Response, next: NextFunction): Promise<void> {
    try {
      const { key } = req.params;
      const { isEnabled, config } = req.body as { isEnabled?: boolean; config?: Record<string, unknown> };

      const setParts: string[] = ['updated_at = NOW()'];
      const params: unknown[] = [];
      let idx = 1;

      if (isEnabled !== undefined) { setParts.push(`is_enabled = $${idx++}`); params.push(isEnabled); }
      if (config !== undefined)    { setParts.push(`config = $${idx++}`);     params.push(JSON.stringify(config)); }
      params.push(key);

      const [updated] = await AppDataSource.query(
        `UPDATE feature_flags SET ${setParts.join(', ')}
         WHERE feature_key = $${idx}
         RETURNING feature_key, entity_type, is_enabled, config, updated_at`,
        params,
      );

      if (!updated) { res.status(404).json({ success: false, error: 'Flag not found' }); return; }
      res.json({ success: true, data: updated });
    } catch (err) { next(err); }
  },
};
```

### 3.1 Service — `src/services/ai-revenue-analytics.service.ts`

```typescript
import { GoogleGenAI } from '@google/genai';
import { AppDataSource } from '../config/database';
import { env } from '../config/env';
import { logger } from '../config/logger';
import { AppError } from '../types';
import * as dashboardService from './provider-dashboard.service';

const ai = new GoogleGenAI({ apiKey: env.ai.googleApiKey });

// 1. Gate — check GLOBAL flag is enabled (admin-owned providers bypass)
export async function checkAnalyticsGate(providerId: string): Promise<{ monthlyQuota: number }> {
  const [userRow] = await AppDataSource.query<{ role: string }[]>(
    `SELECT role FROM users WHERE id = $1`, [providerId],
  );
  if (userRow?.role === 'ADMIN') return { monthlyQuota: 0 }; // unlimited bypass

  const [flag] = await AppDataSource.query<{ is_enabled: boolean; config: { monthly_quota: number } }[]>(
    `SELECT is_enabled, config FROM feature_flags
     WHERE feature_key = 'AI_REVENUE_ANALYTICS' AND entity_type = 'GLOBAL'`,
  );
  if (!flag?.is_enabled) {
    throw new AppError('Tính năng AI Phân Tích Doanh Thu chưa được kích hoạt.', 503, 'AI_ANALYTICS_DISABLED');
  }
  return { monthlyQuota: flag.config?.monthly_quota ?? 10 };
}

// 2. Quota check — count SUCCESS logs for current calendar month
export async function checkAndLogAnalyticsQuota(
  providerId: string, monthlyQuota: number,
  from: string, to: string, cafeId: string | null,
): Promise<string> {
  const logRepo = AppDataSource.getRepository('ai_analysis_logs');

  if (monthlyQuota !== 0) {
    const [{ count }] = await AppDataSource.query<{ count: string }[]>(
      `SELECT COUNT(*) as count FROM ai_analysis_logs
       WHERE provider_id = $1 AND status = 'SUCCESS'
         AND DATE_TRUNC('month', requested_at) = DATE_TRUNC('month', NOW())`,
      [providerId],
    );
    if (parseInt(count, 10) >= monthlyQuota) {
      // Log the blocked attempt
      await AppDataSource.query(
        `INSERT INTO ai_analysis_logs (provider_id, cafe_id, period_from, period_to, status)
         VALUES ($1, $2, $3, $4, 'QUOTA_EXCEEDED')`,
        [providerId, cafeId, from, to],
      );
      throw new AppError(
        `Bạn đã sử dụng hết ${monthlyQuota} lượt phân tích trong tháng này.`,
        429, 'AI_QUOTA_EXCEEDED',
      );
    }
  }

  // Insert pending log row, return its id for later update
  const [{ id }] = await AppDataSource.query<{ id: string }[]>(
    `INSERT INTO ai_analysis_logs (provider_id, cafe_id, period_from, period_to, status)
     VALUES ($1, $2, $3, $4, 'FAILED') RETURNING id`,
    [providerId, cafeId, from, to],
  );
  return id;
}

// 3. Fetch all revenue data in parallel from existing dashboard service functions
export async function fetchRevenueData(
  providerId: string, from: string, to: string, cafeId?: string,
) {
  const [kpi, trend, breakdown, branchPerf, topStats] = await Promise.all([
    dashboardService.getProviderKpi(providerId, from, to, cafeId),
    dashboardService.getProviderRevenueTrend(providerId, 'weekly', from, to, cafeId),
    dashboardService.getProviderRevenueBreakdown(providerId, from, to, cafeId),
    dashboardService.getProviderBranchPerformance(providerId, from, to),
    dashboardService.getProviderTopStats(providerId, from, to, cafeId),
  ]);
  return { kpi, trend, breakdown, branchPerf, topStats };
}

// 4. Compute derived metrics from raw data
export function computeDerivedMetrics(data: Awaited<ReturnType<typeof fetchRevenueData>>) {
  const { kpi, trend, breakdown } = data;
  const completionRate = kpi.totalBookings > 0
    ? (kpi.completedBookings / kpi.totalBookings) * 100 : 0;
  const revenuePerBooking = kpi.completedBookings > 0
    ? kpi.totalRevenue / kpi.completedBookings : 0;

  // Linear slope over weekly trend: positive = rising, negative = falling
  const trendValues = trend.map((t: { value: number }) => t.value);
  const n = trendValues.length;
  let slope = 0;
  if (n >= 2) {
    const meanX = (n - 1) / 2;
    const meanY = trendValues.reduce((a: number, b: number) => a + b, 0) / n;
    const num = trendValues.reduce((acc: number, y: number, i: number) => acc + (i - meanX) * (y - meanY), 0);
    const den = trendValues.reduce((acc: number, _: number, i: number) => acc + (i - meanX) ** 2, 0);
    slope = den !== 0 ? num / den : 0;
  }
  const trendDirection = slope > 0.05 ? 'rising' : slope < -0.05 ? 'falling' : 'flat';

  const topSource = [...breakdown].sort((a: { amount: number }, b: { amount: number }) => b.amount - a.amount)[0];

  return { completionRate, revenuePerBooking, trendDirection, slope, topSource };
}

// 6. Main entry point
export async function generateAiInsights(
  providerId: string, from: string, to: string, cafeId?: string,
) {
  const startMs = Date.now();
  let logId: string | null = null;
  let status: 'SUCCESS' | 'FAILED' | 'INSUFFICIENT_DATA' = 'FAILED';
  let tokensUsed: number | null = null;

  try {
    const { monthlyQuota } = await checkAnalyticsGate(providerId);

    const data = await fetchRevenueData(providerId, from, to, cafeId);

    if (data.kpi.completedBookings === 0) {
      await AppDataSource.query(
        `INSERT INTO ai_analysis_logs (provider_id, cafe_id, period_from, period_to, status, duration_ms)
         VALUES ($1, $2, $3, $4, 'INSUFFICIENT_DATA', $5)`,
        [providerId, cafeId ?? null, from, to, Date.now() - startMs],
      );
      return { type: 'INSUFFICIENT_DATA' as const };
    }

    logId = await checkAndLogAnalyticsQuota(providerId, monthlyQuota, from, to, cafeId ?? null);

    const metrics = computeDerivedMetrics(data);
    const prompt = buildPrompt(data, metrics, from, to);

    const response = await ai.models.generateContent({
      model: env.ai.supportModel,
      contents: [{ role: 'user', parts: [{ text: prompt }] }],
    });

    const text = response.candidates?.[0]?.content?.parts?.[0]?.text ?? '';
    tokensUsed = response.usageMetadata?.totalTokenCount ?? null;
    const parsed = JSON.parse(text);
    status = 'SUCCESS';

    return {
      type: 'SUCCESS' as const,
      data: {
        period: { from, to },
        summary: parsed.summary,
        insights: parsed.insights,
        topOpportunity: parsed.topOpportunity,
        watchouts: parsed.watchouts,
        generatedAt: new Date().toISOString(),
      },
    };
  } catch (err) {
    if ((err as { code?: string }).code === 'AI_QUOTA_EXCEEDED') throw err;
    if ((err as { code?: string }).code === 'AI_ANALYTICS_DISABLED') throw err;
    logger.error('AiAnalytics', 'Gemini call failed', err);
    throw new AppError('Phân tích AI thất bại. Vui lòng thử lại sau.', 503, 'AI_ANALYTICS_FAILED');
  } finally {
    if (logId) {
      await AppDataSource.query(
        `UPDATE ai_analysis_logs
         SET status = $1, tokens_used = $2, duration_ms = $3
         WHERE id = $4`,
        [status, tokensUsed, Date.now() - startMs, logId],
      ).catch(() => {/* log update failure is non-fatal */});
    }
  }
}
```

**Edge case notes**:
- LLM failure → `finally` updates log to `FAILED`; error propagates as 503 without incrementing quota count (log stays FAILED, not counted)
- Zero bookings → logged as `INSUFFICIENT_DATA` before quota check (does not consume quota)
- `QUOTA_EXCEEDED` → logged immediately, throws 429
- JSON parse error from Gemini → caught as FAILED, user sees friendly 503

### 3.2 Controller — `src/controllers/ai-revenue-analytics.controller.ts`

```typescript
import { Response, NextFunction } from 'express';
import { z } from 'zod';
import { AuthRequest } from '../types';
import * as aiAnalyticsService from '../services/ai-revenue-analytics.service';

const QuerySchema = z.object({
  from: z.string().regex(/^\d{4}-\d{2}-\d{2}$/),
  to:   z.string().regex(/^\d{4}-\d{2}-\d{2}$/),
  cafeId: z.string().uuid().optional(),
});

export const aiRevenueAnalyticsController = {
  async generateInsights(req: AuthRequest, res: Response, next: NextFunction): Promise<void> {
    try {
      const query = QuerySchema.parse(req.query);
      const providerId = req.user!.userId;
      const result = await aiAnalyticsService.generateAiInsights(
        providerId, query.from, query.to, query.cafeId,
      );
      if (result.type === 'INSUFFICIENT_DATA') {
        res.json({ success: true, type: 'INSUFFICIENT_DATA', data: null });
        return;
      }
      res.json({ success: true, type: 'SUCCESS', data: result.data });
    } catch (err) { next(err); }
  },
};
```
